package profile

type ID string

const (
	P0 ID = "P0"
	P1 ID = "P1"
	P2 ID = "P2"
	P3 ID = "P3"
)

type ActorBoundary string
type RecoveryObservationRequirement string
type Capability string
type Claim struct{ Identifier, Description string }
type Declaration struct {
	Profile                           ID
	Label                             string
	Guarantees, Nonclaims             []Claim
	OneTargetPerStore, OneRunPerStore bool
	ActorBoundary                     ActorBoundary
	RecoveryObservation               RecoveryObservationRequirement
	RequiredCapabilities              []Capability
	Available                         bool
}
type StoreCapabilities struct{ capabilities []Capability }

func NewStoreCapabilities(capabilities []Capability) (StoreCapabilities, error)
func (s StoreCapabilities) Values() []Capability

var ErrProfileUnavailable error

func DeclarationFor(id ID) (Declaration, error)
func Select(id ID) (Declaration, error)
func RequireCapabilities(declaration Declaration, available []Capability) error
func CoreCapabilities() []Capability
