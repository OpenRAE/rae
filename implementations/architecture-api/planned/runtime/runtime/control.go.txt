	InjectOutcome(context.Context, Identity, string) (*binding.InjectOccurrenceOutcomeV1, error)

	InjectCapture(context.Context, Identity, string) (*InjectCaptureResult, error)

	DispatchParticipantControlEffects(context.Context, Identity, string) ([]EffectRealization, error)
	DispatchParticipantCancellation(context.Context, Identity, string) (ParticipantCancellationStatus, error)
	ParticipantCancellationStatus(context.Context, Identity, string) (ParticipantCancellationStatus, error)
	SuperviseOperation(context.Context, Identity, SupervisionRequest) (binding.BackendOperationControlV1, error)
	SubmitOperationControl(context.Context, Identity, binding.BackendOperationControlV1) (binding.BackendOperationControlV1, error)
	ControlResponse(context.Context, Identity, string, string) (*binding.BackendOperationResponseV1, error)
	StopAdmission(context.Context, Identity) error
	ResolveIndeterminateOperation(context.Context, Identity, ResolutionRequest) (binding.OperationReceiptV1, error)
	OpenRun(context.Context, Identity, Options) (RunContext, error)
	Run(context.Context, Identity, string) (RunContext, error)
	RetireRun(context.Context, Identity, string) error
	AdmitExperiment(context.Context, Identity, ExperimentRequest) (string, error)
	AdvanceExperiment(context.Context, Identity, string) (ExperimentProgress, error)
	CancelExperiment(context.Context, Identity, string) error
	ExperimentResults(context.Context, Identity, string) ([]TrialUnitResult, error)

	AdvanceSequence(context.Context, Identity, execution.RuntimeSequenceReference) (string, error)

	TransitionParticipant(context.Context, Identity, ParticipantTransition) (binding.OperationReceiptV1, error)
	Audit(context.Context, Identity) ([]provider.AuditEvent, error)
	Close() error
}
