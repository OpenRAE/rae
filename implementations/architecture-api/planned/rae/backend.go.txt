	Reset(context.Context, runtimecontract.EpisodeResetRequest, binding.RuntimeSnapshotV1) (runtimecontract.ApplyResult, error)

type TimeRuntime interface {
	Initialize(context.Context, binding.TimeModelV1, binding.RuntimeSnapshotV1) (runtimecontract.ApplyResult, error)
	Advance(context.Context, string, int64, int64, binding.RuntimeSnapshotV1) (runtimecontract.ApplyResult, error)

	Resume(context.Context, string, binding.RuntimeSnapshotV1) (runtimecontract.ApplyResult, error)
	Jump(context.Context, string, int64, int64, binding.RuntimeSnapshotV1) (runtimecontract.ApplyResult, error)
	Reset(context.Context, string, bool, binding.RuntimeSnapshotV1) (runtimecontract.ApplyResult, error)
