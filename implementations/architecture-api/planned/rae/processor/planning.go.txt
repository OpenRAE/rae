package planning

import (
	"github.com/OpenRAE/rae/implementations/go/contract/admission"
	"github.com/OpenRAE/rae/implementations/go/contract/binding"
	"github.com/OpenRAE/rae/implementations/go/contract/diagnostic"
	"github.com/OpenRAE/rae/implementations/go/contract/validation"
	"github.com/OpenRAE/rae/implementations/go/processor"
)

type Result struct {
	Plan        ExecutionPlan
	Resources   map[string]int
	Diagnostics []diagnostic.Diagnostic
}

// Only the planner can issue this handle. Projections return copies.
type ExecutionPlan interface {
	Valid() bool
	Model() processor.RuntimeModel
	Manifest() binding.BackendManifestV2
	Provisioning() binding.ProvisioningPlanV1
	Orchestration() binding.OrchestrationPlanV1
	Evaluation() binding.EvaluationPlanV1
	Context() PlanningContext
	plannerProduced()
}
type Scope struct{ TargetName, RunID, InstantiationID *string }
type Closure string

const (
	OpenWorld   Closure = "open-world"
	ClosedWorld Closure = "closed-world"
)

type ApparatusDefaultResolver interface {
	Resolve(admission.Fragment, binding.BackendManifestV2) (Closure, error)
}
type ArtifactAvailability struct {
	Address                                                                                                              string
	AvailableArtifactDigests, AvailableCandidateIDs, VerifiedLockedInputIDs, SatisfiedConstraintIDs                      []string
	AvailableMaterializationSpecificationDigests                                                                         []string
	VerifiedIntegrityRefs, VerifiedAuthenticityRefs, VerifiedAdmissionRefs, VerifiedProvenanceRefs, VerifiedEvidenceRefs []string
}
type MixedControlAdmission interface {
	Model() processor.RuntimeModel
	mixedControlAdmission()
}

func NewMixedControlAdmission(processor.RuntimeModel) (MixedControlAdmission, error)

type Options struct {
	Snapshot              *binding.RuntimeSnapshotV1
	Scope                 *Scope
	ApparatusDefault      ApparatusDefaultResolver
	ArtifactAvailability  []ArtifactAvailability
	ProfileContext        *binding.DomainProfileResolutionContextV1
	MixedControlAdmission MixedControlAdmission
	RefreshEvaluation     bool
	ProcessorManifest     *binding.ProcessorManifestV2
}
type RequirementKey struct{ Address, FieldPath, RequirementKind string }
type PlanningContext struct {
	Snapshot              binding.RuntimeSnapshotV1
	Scope                 *Scope
	ApparatusDecisions    map[RequirementKey]Closure
	ArtifactAvailability  []ArtifactAvailability
	ProfileContext        *binding.DomainProfileResolutionContextV1
	MixedControlAdmission MixedControlAdmission
	RefreshEvaluation     bool
	ProcessorManifest     *binding.ProcessorManifestV2
}

func Plan(compilation processor.Compilation, manifest validation.Value, options Options) (Result, error)
func PlanWithManifest(compilation processor.Compilation, manifest binding.BackendManifestV2, options Options) (Result, error)
