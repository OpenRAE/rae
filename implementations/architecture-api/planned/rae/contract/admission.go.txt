// Fragment is an admitted native submodel, not a published document Value.
// Profile identifies the exact existing owner model; Document returns a copy.
type Fragment interface {
	Profile() FragmentProfile
	Document() any

}

type ImplementationIdentity struct{ Name, Version string }
type BindingOwner struct{ ContractID, ContractVersion, ValidatorID, ValidatorVersion string }

// Literal is a strict JSON scalar; ReferenceID is permitted only for secret-reference.
type BindingValue struct {
	Kind        string
	Literal     any
	ReferenceID string
}
type ControlContextAssetReference struct{ AssetID, AssetVersion, AssetDigest string }
type ControlContextSelection struct {
	SlotID, DeclarationRef string
	AssetRef               ControlContextAssetReference
}
type ConfigurationOverride struct {
	TargetID string
	Value    BindingValue
}
type RealizedConfigurationValue struct {
	TargetID, ValueType, Origin string
	Value                       BindingValue
	ControlContextSelection     *ControlContextSelection
}
type ParticipantConfiguration struct {
	ImplementationIdentity ImplementationIdentity
	ManifestVersion        string
	Owner                  BindingOwner
	Values                 []RealizedConfigurationValue
}
type ParticipantConfigurationValidator interface {
	ValidateAndNormalize(context.Context, ParticipantConfiguration) (ParticipantConfiguration, error)
}
type ParticipantControlContextValidator interface {
	ValidateControlContextReference(context.Context, string, ControlContextSelection) error
}
type ParticipantConfigurationRequest struct {
	ParticipantAddress, ManifestRef, ManifestDigest string
	Manifest                                        binding.ParticipantImplementationManifestV1
	Overrides                                       []ConfigurationOverride
	Validator                                       ParticipantConfigurationValidator
	ControlContext                                  ParticipantControlContextValidator
}

func RealizeParticipantConfiguration(context.Context, ParticipantConfigurationRequest) (binding.ParticipantConfigurationResultV1, error)
func ValidateParticipantConfigurationSelection(Fragment, binding.ParticipantConfigurationResultV1) error

type ParticipantManifestKey struct{ ParticipantAddress, ImplementationName, ImplementationVersion, ManifestVersion string }
type ApparatusManifestKey struct{ ComponentKind, ComponentName, ComponentVersion, ManifestVersion string }
type ApparatusManifest struct {
	Processor *binding.ProcessorManifestV2
	Backend   *binding.BackendManifestV2
}
type ScenarioBindingResolution struct {
	CanonicalTargetID, ValueType, Sensitivity string
	AllowedValueKinds                         []string
	Owner                                     BindingOwner
}
type ScenarioBindingTargetResolver interface {
	Resolve(scenarioFamilyID, variationPointID, suppliedTargetID string) (ScenarioBindingResolution, error)
}

func ValidateExperimentBindingTargets(descriptors binding.ExperimentBindingDescriptorsV1, resolver ScenarioBindingTargetResolver, participants map[ParticipantManifestKey]binding.ParticipantImplementationManifestV1, apparatus map[ApparatusManifestKey]ApparatusManifest) (binding.ExperimentBindingDescriptorsV1, error)
func MaterializationArchive(v *Validator, record binding.MaterializationArchiveRecordV1, inventory *binding.BackendRuntimeDescendantsV1, limits ingress.Limits, context MaterializationInventoryContext) (validation.Value, error)
func RestoreMaterializationArchive(v *Validator, data []byte, limits ingress.Limits, context MaterializationInventoryContext) (binding.MaterializationArchiveRecordV1, *binding.BackendRuntimeDescendantsV1, validation.Value, error)
